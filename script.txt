/*

SECTION 2: Generate Monthly Allocation Factors
================================================================================
Create 12 random numbers per FactID, sum them, then normalize to AllocationFactor
so each FactID's 12 factors sum to 1.
*/
Monthly_Randoms_CC:

    FactID,
    IterNo() as Month,
    Rand() as RandomValue
RESIDENT Facts_Temp_CC

    FactID,
    Month,
    RandomValue / TotalRandomValue as AllocationFactor
RESIDENT Monthly_Randoms_CC;

///$tab FACT - Balance Sheet distributed
/*
================================================================================
SECTION 1: Initial Data Load
================================================================================
First, we load the original data from the QVD file.
We add two new fields:
- FactID: A unique row number (RowNo()) for each original record. This is crucial for linking the random allocations back to the correct original value.
- Year: We extract the year from the original date. This will be used to construct the new monthly dates (01.01.YYYY, 01.02.YYYY, etc.).
*/
Facts_Temp:
LOAD
	RowNo() as FactID,
    'Jedox' as Datasource,
    'Balance Sheet' as Datatype,
    makedate(left("Month",4),month(right(Month,2)),1) as OriginalDate,
    Year(makedate(left("Month",4),month(right(Month,2)),1)) as Year,
    Version as Scenario,
    'Currency' as Unittype,
    'EUR' as Unit,
    'Month' as "Aggregation Level",
    text("Legal Entity") as "Legal Entity ID",
	text("Balance Sheet Account") as "Account ID",
    Currency,
    "Transaction Type" as "Transaction Type",
    Value
FROM [lib://01_Development:DataFiles/2020/Facts Balance Sheet.qvd]
(qvd);

/*
================================================================================
SECTION 2: Generate Monthly Allocation Factors
================================================================================
This section contains the core logic for the random distribution.
It's a three-step process to ensure the allocation is both random and adds up to 1 (100%).

STEP 2a: For each original record (FactID), create 12 new rows (one for each month)
and assign a random number to each.
*/
Monthly_Randoms:
LOAD
    FactID,
    IterNo() as Month,
    Rand() as RandomValue
RESIDENT Facts_Temp
WHILE IterNo() <= 12;

/*
STEP 2b: Calculate the sum of the 12 random numbers for each original record.
We then join this total back to our monthly random numbers table.
*/
LEFT JOIN (Monthly_Randoms)
LOAD
    FactID,
    Sum(RandomValue) as TotalRandomValue
RESIDENT Monthly_Randoms
GROUP BY FactID;

/*
STEP 2c: Calculate the final allocation factor for each month by dividing its
random number by the total. This normalizes the values, so for any given FactID,
the sum of the 12 AllocationFactor values will equal 1.
*/
Allocation_Factors:
NOCONCATENATE
LOAD
    FactID,
    Month,
    RandomValue / TotalRandomValue as AllocationFactor
RESIDENT Monthly_Randoms;

// We no longer need this intermediate table, so we drop it.
DROP TABLE Monthly_Randoms;

/*
================================================================================
SECTION 3: Distribute Values and Create the Final Table
================================================================================
STEP 3a: Create an intermediate table containing the original facts.
*/
Facts_Distributed_Temp:
NOCONCATENATE
LOAD
    FactID,
    Datasource,
    Datatype,
    Year,
    Scenario,
    Unittype,
    Unit,
    "Aggregation Level",
    "Legal Entity ID",
    "Account ID",
    Currency,
    "Transaction Type",
    Value
RESIDENT Facts_Temp;

/*
STEP 3b: Join the monthly allocation factors to the intermediate table.
Qlik will automatically join on the common field 'FactID'.
*/
JOIN (Facts_Distributed_Temp)
LOAD
    FactID,
    Month,
    AllocationFactor
RESIDENT Allocation_Factors;

/*
STEP 3c: Create the final table from the joined intermediate table.
Here we calculate the new distributed 'Value' and the new 'Date', and
ensure we only keep the final fields needed.
*/
Facts:
CONCATENATE (Facts)
LOAD
    Datasource,
    Datatype,
    MakeDate(Year, Month, 1) as Date,
    Scenario,
    Unittype,
    Unit,
    "Aggregation Level",
    "Legal Entity ID",
    "Account ID",
    Currency,
    "Transaction Type",
    Value * AllocationFactor as Value
RESIDENT Facts_Distributed_Temp;

// Drop the intermediate table as it's no longer needed.
DROP TABLE Facts_Distributed_Temp;


/*
================================================================================
SECTION 4: Cleanup
================================================================================
Finally, we drop the other temporary tables used during the script execution,
leaving only the final 'Facts_Distributed' table in the data model.
*/
DROP TABLE Facts_Temp;
DROP TABLE Allocation_Factors;

///$tab DIM - Structure Link
/********************************************************************************
* Script to Create a Flat Hierarchy Link Table
*
* Purpose: This script transforms a hierarchical 'Accounts' table (with levels
* in separate columns like Level1, Level2, etc.) into a flat link table.
* This new table allows for easy selection and analysis of all lowest-level
* accounts that roll up to any given parent at any level.
*********************************************************************************/

// Step 1: Initialize the target link table.
// This is a best practice to ensure the Concatenate function works correctly,
// even if the first iteration of the loop finds no data.
"Structure Link":
LOAD 
'' as "Account ID"
autogenerate(0);


// Step 2: Loop through each of the 5 hierarchy levels to flatten the structure.
FOR i = 1 TO 11

    // Create a variable to hold the name of the current level field (e.g., 'Level1', 'Level2', etc.)
    LET vLevelField = 'Level' & $(i);

    // In each loop, load the links between the parent at the current level and all
    // its child accounts. Concatenate the results into the final table.
    Concatenate ("Structure Link")
    LOAD
        // Create the composite key: "Dimension/Level Number/Structure Text"
        // This key uniquely identifies the relationship.
        Dimension & '/' & '$(i)' & '/' & [$(vLevelField)] as "Structure Link",

        // Load the core fields for the flat link table
//         Dimension,
//         '$(i)' as "Level Number",
//         [$(vLevelField)] as "Structure Text",

        // These fields represent the lowest-level member of the hierarchy
        "Account ID"
//         "Account Name"

    RESIDENT Accounts
    // This WHERE clause is crucial. It ensures we only create links where a
    // parent value actually exists for that level, preventing blank/null links.
    WHERE Len(Trim([$(vLevelField)])) > 0;

NEXT i;


/********************************************************************************
* End of Script
*
* The resulting "Structure Link" now contains the flattened structure.
*
* Example usage in the front-end:
* - A filter on "Level Number" = 3
* - A filter on "Structure Text" = 'Total revenue'
* - A table box showing the resulting "Account Name" values.
*********************************************************************************/

///$tab FACT - P&L distibuted (original)
/*
================================================================================
SECTION 1: Initial Data Load
================================================================================
First, we load the original data from the QVD file.
We add two new fields:
- FactID: A unique row number (RowNo()) for each original record. This is crucial for linking the random allocations back to the correct original value.
- Year: We extract the year from the original date. This will be used to construct the new monthly dates (01.01.YYYY, 01.02.YYYY, etc.).
*/
Facts_Temp:
LOAD
    RowNo() as FactID,
	'Jedox' as Datasource,
    'P&L' as Datatype,
    makedate(left("Month",4),month(right(Month,2)),1) as OriginalDate,
    Year(makedate(left("Month",4),month(right(Month,2)),1)) as Year,
    Version as Scenario,
    'Currency' as Unittype,
    'EUR' as Unit,
    'Month' as "Aggregation Level",
    text("Legal Entity") as "Legal Entity ID",
    text("Profit and Loss Account") as "Account ID",
    Currency,
    if(Version='Actual',Value * 3.5, 
     	if(wildmatch(Version,'*Forecast*'),Value * 1.5,Value)) as Value
FROM [lib://01_Development:DataFiles/2020/Facts Profit and Loss.qvd]
(qvd);

/*
================================================================================
SECTION 2: Generate Monthly Allocation Factors
================================================================================
This section contains the core logic for the random distribution.
It's a three-step process to ensure the allocation is both random and adds up to 1 (100%).

STEP 2a: For each original record (FactID), create 12 new rows (one for each month)
and assign a random number to each.
*/
Monthly_Randoms:
LOAD
    FactID,
    IterNo() as Month,
    Rand() as RandomValue
RESIDENT Facts_Temp
WHILE IterNo() <= 12;

/*
STEP 2b: Calculate the sum of the 12 random numbers for each original record.
We then join this total back to our monthly random numbers table.
*/
LEFT JOIN (Monthly_Randoms)
LOAD
    FactID,
    Sum(RandomValue) as TotalRandomValue
RESIDENT Monthly_Randoms
GROUP BY FactID;

/*
STEP 2c: Calculate the final allocation factor for each month by dividing its
random number by the total. This normalizes the values, so for any given FactID,
the sum of the 12 AllocationFactor values will equal 1.
*/
Allocation_Factors:
NOCONCATENATE
LOAD
    FactID,
    Month,
    RandomValue / TotalRandomValue as AllocationFactor
RESIDENT Monthly_Randoms;

// We no longer need this intermediate table, so we drop it.
DROP TABLE Monthly_Randoms;

/*
================================================================================
SECTION 3: Distribute Values and Create the Final Table
================================================================================
STEP 3a: Create an intermediate table containing the original facts.
*/
Facts_Distributed_Temp:
NOCONCATENATE
LOAD
    FactID,
    Datasource,
    Datatype,
    Year,
    Scenario,
    Unittype,
    Unit,
    "Aggregation Level",
    "Legal Entity ID",
    "Account ID",
    Currency,
    Value
RESIDENT Facts_Temp;

/*
STEP 3b: Join the monthly allocation factors to the intermediate table.
Qlik will automatically join on the common field 'FactID'.
*/
JOIN (Facts_Distributed_Temp)
LOAD
    FactID,
    Month,
    AllocationFactor
RESIDENT Allocation_Factors;

/*
STEP 3c: Create the final table from the joined intermediate table.
Here we calculate the new distributed 'Value' and the new 'Date', and
ensure we only keep the final fields needed.
*/
Facts:
NOCONCATENATE
LOAD
    Datasource,
    Datatype,
    MakeDate(Year, Month, 1) as Date,
    Scenario,
    Unittype,
    Unit,
    "Aggregation Level",
    "Legal Entity ID",
    "Account ID",
    Currency,
    Value * AllocationFactor as Value
RESIDENT Facts_Distributed_Temp;

// Drop the intermediate table as it's no longer needed.
DROP TABLE Facts_Distributed_Temp;


/*
================================================================================
SECTION 4: Cleanup
================================================================================
Finally, we drop the other temporary tables used during the script execution,
leaving only the final 'Facts_Distributed' table in the data model.
*/
DROP TABLE Facts_Temp;
DROP TABLE Allocation_Factors;
